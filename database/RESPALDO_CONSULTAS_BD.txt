QUE HACE:
VALIDA EL ID_TP Y SELECCIONA LA TABLA CORRECTA SEGUN LA FORMA DE PAGO.
ID_TP 1 INSERTA EN dbo.VENTA_EFECTIVO.
ID_TP 2 INSERTA EN dbo.VENTA_CREDITO.
ID_TP 3 INSERTA EN dbo.VENTA_TRANSFERENCIA.
INSERTA CADA LINEA DE LA VENTA EN LA TABLA CORRECTA.
GUARDA ID_TP CON EL ID DE dbo.TIPO_PAGO Y GUARDA ID_CLIENTE.
SI LA VENTA ES CREDITO, ID_CLIENTE DEBE VENIR DEL CLIENTE SELECCIONADO.

GUARDA ID_ESTADO_VENTA SEGUN LA FORMA DE PAGO.
EFECTIVO Y TRANSFERENCIA GUARDAN 4 PORQUE LA VENTA QUEDA FINALIZADA.
CREDITO GUARDA 2 PORQUE QUEDA ACTIVA/PENDIENTE.
NO ACTUALIZA INVENTARIO.

QUE HACE:
OBTIENE LOS CLIENTES DESDE LA TABLA cliente
PARA LLENAR EL MODAL DE SELECCION DE CLIENTES EN FACTURACION.
CALCULA EL SALDO DESDE dbo.VENTA_CREDITO SUMANDO PRECIO_VENTA * CANT_PD
DE CREDITOS ACTIVOS CON ID_ESTADO_VENTA = 2.

CONSULTA:
SELECT
  c.ID_CLIENTE,
  c.NOMBRE,
  c.APELLIDO,
  c.TELEFONO,
  c.DIRECCION,
  c.CREDITOS_ABIERTOS,
  c.FECHA_HORA,
  ISNULL(creditos.SALDO_CREDITO, 0) AS saldo
FROM dbo.cliente c
LEFT JOIN (
  SELECT
    ID_CLIENTE,
    SUM(PRECIO_VENTA * CANT_PD) AS SALDO_CREDITO
  FROM dbo.VENTA_CREDITO
  WHERE ID_ESTADO_VENTA = 2
  GROUP BY ID_CLIENTE
) creditos
  ON creditos.ID_CLIENTE = c.ID_CLIENTE
ORDER BY c.ID_CLIENTE ASC

QUE HACEN:
SE AGREGARON DOS SELECTORES AL FORMULARIO DE VENTA.
UNO PERMITE ELEGIR LA FORMA DE PAGO ENTRE EFECTIVO, CREDITO Y TRANSFERENCIA.
OTRO PERMITE SELECCIONAR UN CLIENTE PARA AGREGARLO A LA FACTURA.
EL LISTADO DE CLIENTES AHORA SE CARGA DESDE LA TABLA cliente.
VISUALMENTE LAS FORMAS DE PAGO SE MUESTRAN COMO BOTONES SEGMENTADOS
EN LA NAV DEL FORMULARIO DE VENTA.
LOS BOTONES DE FORMA DE PAGO SOLO MUESTRAN LAS INICIALES.
SE APLICO FONDO DEGRADADO PARA QUE TENGAN MEJOR PRESENCIA VISUAL.
CUANDO UN BOTON ESTA SELECCIONADO USA UN DEGRADADO ENTRE MORADO Y AZUL.
EL CLIENTE QUEDO COMO UN SOLO BOTON QUE ABRE UN MODAL
CON UNA TABLA TEMPORAL DE CLIENTES PARA SELECCIONAR UNO.
EL MODAL SE MUESTRA CENTRADO EN LA PAGINA Y EL FONDO USA BLUR
PARA QUE EL CONTENIDO POSTERIOR SE VEA DIFUMINADO Y RESALTE EL MODAL.
EL MODAL AHORA QUEDA DENTRO DEL CONTENEDOR PRINCIPAL PARA HEREDAR
EL TEMA ACTIVO QUE ESTE USANDO LA APLICACION.
CUANDO SE REGISTRA UNA VENTA, paymentMethodOptions ENVIA EL ID_TP
DE dbo.TIPO_PAGO HACIA persistSale() Y requestCreateSale().
LOS IDS ACTUALES SON:
1 = EFECTIVO
2 = CREDITO
3 = TRANSFERENCIA
selectCustomer(customer) GUARDA EL ID_CLIENTE SELECCIONADO PARA
ENVIARLO EN requestCreateSale() Y QUE registerSale() LO INSERTE
EN LA TABLA DE VENTA CORRECTA SEGUN ID_TP.
ANTES DE GUARDAR CUALQUIER VENTA, persistSale() VALIDA QUE EXISTA
UN CLIENTE SELECCIONADO PARA PODER ENVIAR ID_CLIENTE.

  vc.ID_PD,
  p.nombre AS PRODUCTO,
  vc.CANT_PD,
  vc.PRECIO_COSTO,
  vc.PRECIO_VENTA,

  vc.USUARIO,
  vc.ID_CLIENTE,
  c.NOMBRE,
  c.APELLIDO,
  c.TELEFONO,
  c.CREDITOS_ABIERTOS,
  c.saldo,
  vc.ID_TP,

RESPALDO DE CONSULTA PARA TARJETAS DE VENTAS Y COMPRAS DEL DASHBOARD

UBICACION DEL CODIGO:
server/data-access.js

PROCEDIMIENTO QUE LA UTILIZA:
getDashboardSalesSummary()

QUE HACE:
CONSULTA LOS TOTALES DE VENTAS Y COMPRAS PARA LAS TARJETAS DEL DASHBOARD.
LA PRIMERA TARJETA DE EFECTIVO + TRANSFERENCIAS TIENE SELECTOR PARA MOSTRAR
EL MES ACTUAL O EL HISTORICO ACUMULADO.
LA SEGUNDA TARJETA DE VENTAS AL CREDITO TIENE SELECTOR PARA MOSTRAR
EL MES ACTUAL O EL HISTORICO ACUMULADO.
LA TERCERA TARJETA DE VENTAS TOTALES TIENE SELECTOR PARA MOSTRAR
EL MES ACTUAL O EL HISTORICO ACUMULADO.
LA CUARTA TARJETA DE COMPRAS TOTALES TIENE SELECTOR PARA MOSTRAR
EL MES ACTUAL O EL HISTORICO ACUMULADO.
LOS DOS PRIMEROS TOTALES ACUMULADOS SE CALCULAN MULTIPLICANDO
PRECIO_VENTA * CANT_PD EN CADA TABLA.
LOS DOS PRIMEROS TOTALES MENSUALES FILTRAN SOLO EL MES ACTUAL.
EL TOTAL DEL MES SE TOMA DESDE dbo.VENTA_EFECTIVO, dbo.VENTA_CREDITO
Y dbo.VENTA_TRANSFERENCIA SUMANDO PRECIO_VENTA * CANT_PD.
EL TOTAL ACUMULADO DE VENTAS SE TOMA DESDE ESAS MISMAS TRES TABLAS
SIN FILTRO DE FECHA.
EL TOTAL DE COMPRAS SE TOMA DESDE dbo.COMPRA_EFECTIVO Y dbo.COMPRA_CREDITO
SUMANDO PRECIO_COSTO * CANT_PD Y FILTRANDO EL MES ACTUAL.
EL TOTAL ACUMULADO DE COMPRAS SE TOMA DESDE ESAS DOS TABLAS
SIN FILTRO DE FECHA.

CONSULTA:
SELECT
  ISNULL((
    SELECT SUM(PRECIO_VENTA * CANT_PD)
    FROM dbo.VENTA_EFECTIVO
  ), 0) +
  ISNULL((
    SELECT SUM(PRECIO_VENTA * CANT_PD)
    FROM dbo.VENTA_TRANSFERENCIA
  ), 0) AS cash_transfer_total,
  ISNULL((
    SELECT SUM(total)
    FROM (
      SELECT CAST(PRECIO_VENTA * CANT_PD AS decimal(18, 2)) AS total, TRY_CONVERT(datetime2, FECHA_HORA) AS sale_date
      FROM dbo.VENTA_EFECTIVO
      WHERE TRY_CONVERT(datetime2, FECHA_HORA) IS NOT NULL

      UNION ALL

      SELECT CAST(PRECIO_VENTA * CANT_PD AS decimal(18, 2)) AS total, TRY_CONVERT(datetime2, FECHA_HORA) AS sale_date
      FROM dbo.VENTA_TRANSFERENCIA
      WHERE TRY_CONVERT(datetime2, FECHA_HORA) IS NOT NULL
    ) current_month_cash_transfer
    WHERE sale_date >= DATEADD(MONTH, DATEDIFF(MONTH, 0, GETDATE()), 0)
      AND sale_date < DATEADD(MONTH, DATEDIFF(MONTH, 0, GETDATE()) + 1, 0)
  ), 0) AS current_month_cash_transfer_total,
  ISNULL((
    SELECT SUM(PRECIO_VENTA * CANT_PD)
    FROM dbo.VENTA_CREDITO
  ), 0) AS credit_total,
  ISNULL((
    SELECT SUM(PRECIO_VENTA * CANT_PD)
    FROM dbo.VENTA_CREDITO
    WHERE TRY_CONVERT(datetime2, FECHA_HORA) >= DATEADD(MONTH, DATEDIFF(MONTH, 0, GETDATE()), 0)
      AND TRY_CONVERT(datetime2, FECHA_HORA) < DATEADD(MONTH, DATEDIFF(MONTH, 0, GETDATE()) + 1, 0)
  ), 0) AS current_month_credit_total,
  ISNULL((
    SELECT SUM(total)
    FROM (
      SELECT CAST(PRECIO_VENTA * CANT_PD AS decimal(18, 2)) AS total, TRY_CONVERT(datetime2, FECHA_HORA) AS sale_date
      FROM dbo.VENTA_EFECTIVO
      WHERE TRY_CONVERT(datetime2, FECHA_HORA) IS NOT NULL

      UNION ALL

      SELECT CAST(PRECIO_VENTA * CANT_PD AS decimal(18, 2)) AS total, TRY_CONVERT(datetime2, FECHA_HORA) AS sale_date
      FROM dbo.VENTA_CREDITO
      WHERE TRY_CONVERT(datetime2, FECHA_HORA) IS NOT NULL

      UNION ALL

      SELECT CAST(PRECIO_VENTA * CANT_PD AS decimal(18, 2)) AS total, TRY_CONVERT(datetime2, FECHA_HORA) AS sale_date
      FROM dbo.VENTA_TRANSFERENCIA
      WHERE TRY_CONVERT(datetime2, FECHA_HORA) IS NOT NULL
    ) current_month_sales
    WHERE sale_date >= DATEADD(MONTH, DATEDIFF(MONTH, 0, GETDATE()), 0)
      AND sale_date < DATEADD(MONTH, DATEDIFF(MONTH, 0, GETDATE()) + 1, 0)
  ), 0) AS current_month_sales_total,
  ISNULL((
    SELECT SUM(total)
    FROM (
      SELECT CAST(PRECIO_VENTA * CANT_PD AS decimal(18, 2)) AS total
      FROM dbo.VENTA_EFECTIVO

      UNION ALL

      SELECT CAST(PRECIO_VENTA * CANT_PD AS decimal(18, 2)) AS total
      FROM dbo.VENTA_CREDITO

      UNION ALL

      SELECT CAST(PRECIO_VENTA * CANT_PD AS decimal(18, 2)) AS total
      FROM dbo.VENTA_TRANSFERENCIA
    ) accumulated_sales
  ), 0) AS accumulated_sales_total,
  ISNULL((
    SELECT SUM(total)
    FROM (
      SELECT
        CAST(PRECIO_COSTO * CANT_PD AS decimal(18, 2)) AS total,
        TRY_CONVERT(datetime2, FECHA_HORA) AS purchase_date
      FROM dbo.COMPRA_EFECTIVO
      WHERE TRY_CONVERT(datetime2, FECHA_HORA) IS NOT NULL

      UNION ALL

      SELECT
        CAST(PRECIO_COSTO * CANT_PD AS decimal(18, 2)) AS total,
        TRY_CONVERT(datetime2, FECHA_HORA) AS purchase_date
      FROM dbo.COMPRA_CREDITO
      WHERE TRY_CONVERT(datetime2, FECHA_HORA) IS NOT NULL
    ) current_month_purchases
    WHERE purchase_date >= DATEADD(MONTH, DATEDIFF(MONTH, 0, GETDATE()), 0)
      AND purchase_date < DATEADD(MONTH, DATEDIFF(MONTH, 0, GETDATE()) + 1, 0)
  ), 0) AS current_month_purchases_total,
  ISNULL((
    SELECT SUM(total)
    FROM (
      SELECT CAST(PRECIO_COSTO * CANT_PD AS decimal(18, 2)) AS total
      FROM dbo.COMPRA_EFECTIVO

      UNION ALL

      SELECT CAST(PRECIO_COSTO * CANT_PD AS decimal(18, 2)) AS total
      FROM dbo.COMPRA_CREDITO
    ) accumulated_purchases
  ), 0) AS accumulated_purchases_total

RESPALDO DE CONSULTA PARA GRAFICO DE TENDENCIA DE VENTAS DEL DASHBOARD

UBICACION DEL CODIGO:
server/data-access.js

PROCEDIMIENTO QUE LA UTILIZA:
getDashboardSalesTrend(period)

QUE HACE:
CONSULTA LAS VENTAS REALES PARA LLENAR EL GRAFICO DE TENDENCIA DE VENTAS
EN LA PAGINA DASHBOARD.
UNE LAS TRES TABLAS DE VENTA: dbo.VENTA_EFECTIVO, dbo.VENTA_CREDITO
Y dbo.VENTA_TRANSFERENCIA.
AGRUPA LOS DATOS POR DIA, SEMANA O MES SEGUN EL SELECTOR DEL DASHBOARD.
DEVUELVE LOS TOTALES SEPARADOS POR FORMA DE PAGO PARA DIBUJAR TRES LINEAS:
EFECTIVO, CREDITO Y TRANSFERENCIA.

CONSULTA:
WITH ventas AS (
  SELECT
    TRY_CONVERT(datetime2, FECHA_HORA) AS sale_date,
    CAST(PRECIO_VENTA * CANT_PD AS decimal(18, 2)) AS total,
    'efectivo' AS payment_type
  FROM dbo.VENTA_EFECTIVO
  WHERE TRY_CONVERT(datetime2, FECHA_HORA) IS NOT NULL

  UNION ALL

  SELECT
    TRY_CONVERT(datetime2, FECHA_HORA) AS sale_date,
    CAST(PRECIO_VENTA * CANT_PD AS decimal(18, 2)) AS total,
    'credito' AS payment_type
  FROM dbo.VENTA_CREDITO
  WHERE TRY_CONVERT(datetime2, FECHA_HORA) IS NOT NULL

  UNION ALL

  SELECT
    TRY_CONVERT(datetime2, FECHA_HORA) AS sale_date,
    CAST(PRECIO_VENTA * CANT_PD AS decimal(18, 2)) AS total,
    'transferencia' AS payment_type
  FROM dbo.VENTA_TRANSFERENCIA
  WHERE TRY_CONVERT(datetime2, FECHA_HORA) IS NOT NULL
)
SELECT *
FROM (
  SELECT TOP 12
    [EXPRESION DE FECHA SEGUN PERIODO] AS period_start,
    SUM(CASE WHEN payment_type = 'efectivo' THEN total ELSE 0 END) AS efectivo_total,
    SUM(CASE WHEN payment_type = 'credito' THEN total ELSE 0 END) AS credito_total,
    SUM(CASE WHEN payment_type = 'transferencia' THEN total ELSE 0 END) AS transferencia_total
  FROM ventas
  GROUP BY [EXPRESION DE FECHA SEGUN PERIODO]
  ORDER BY period_start DESC
) trend
ORDER BY period_start ASC

RESPALDO DE CONSULTA PARA PAGINA FACTURAS - LISTADO MASTER

UBICACION DEL CODIGO:
server/data-access.js

PROCEDIMIENTO QUE LA UTILIZA:
listInvoices()

QUE HACE:
CONSULTA TODAS LAS FACTURAS GENERADAS DESDE dbo.FACTURA.
HACE JOIN CON dbo.cliente Y dbo.usuario PARA MOSTRAR CLIENTE Y USUARIO.
UNE dbo.VENTA_EFECTIVO, dbo.VENTA_CREDITO Y dbo.VENTA_TRANSFERENCIA
PARA CALCULAR TOTAL, UTILIDAD, CANTIDAD DE LINEAS Y ESTADO DE LA FACTURA.
SI TODAS LAS LINEAS ESTAN EN ID_ESTADO_VENTA = 3, LA FACTURA SE MUESTRA ANULADA.

CONSULTA:
WITH lineas AS (
  SELECT ID_FACT, ID_ESTADO_VENTA, PRECIO_VENTA * CANT_PD AS TOTAL, UTILIDAD, CANT_PD
  FROM dbo.VENTA_EFECTIVO
  WHERE ID_FACT IS NOT NULL

  UNION ALL

  SELECT ID_FACT, ID_ESTADO_VENTA, PRECIO_VENTA * CANT_PD AS TOTAL, UTILIDAD, CANT_PD
  FROM dbo.VENTA_CREDITO
  WHERE ID_FACT IS NOT NULL

  UNION ALL

  SELECT ID_FACT, ID_ESTADO_VENTA, PRECIO_VENTA * CANT_PD AS TOTAL, UTILIDAD, CANT_PD
  FROM dbo.VENTA_TRANSFERENCIA
  WHERE ID_FACT IS NOT NULL
)
SELECT
  f.ID_FACT,
  f.ID_CLIENTE,
  CONCAT(ISNULL(c.NOMBRE, ''), ' ', ISNULL(c.APELLIDO, '')) AS CLIENTE,
  c.TELEFONO,
  f.CANT_ART,
  f.ID_USUARIO,
  u.nombre AS USUARIO,
  f.SUBTOTAL,
  f.FECHA_HORA,
  f.ID_TP,
  CASE f.ID_TP
    WHEN 1 THEN 'Efectivo'
    WHEN 2 THEN 'Credito'
    WHEN 3 THEN 'Transferencia'
    ELSE 'Sin tipo'
  END AS TIPO_PAGO,
  COUNT(lineas.ID_FACT) AS LINEAS,
  ISNULL(SUM(lineas.TOTAL), 0) AS TOTAL_LINEAS,
  ISNULL(SUM(lineas.UTILIDAD), 0) AS UTILIDAD_TOTAL,
  ISNULL(SUM(lineas.CANT_PD), 0) AS CANTIDAD_TOTAL,
  SUM(CASE WHEN lineas.ID_ESTADO_VENTA = 3 THEN 1 ELSE 0 END) AS LINEAS_ANULADAS,
  CASE
    WHEN COUNT(lineas.ID_FACT) > 0
      AND COUNT(lineas.ID_FACT) = SUM(CASE WHEN lineas.ID_ESTADO_VENTA = 3 THEN 1 ELSE 0 END)
      THEN 'Anulado'
    WHEN f.ID_TP = 2 THEN 'Activo'
    ELSE 'Cerrado'
  END AS ESTADO_FACTURA
FROM dbo.FACTURA f
LEFT JOIN dbo.cliente c
  ON c.ID_CLIENTE = f.ID_CLIENTE
LEFT JOIN dbo.usuario u
  ON u.id_usuario = f.ID_USUARIO
LEFT JOIN lineas
  ON lineas.ID_FACT = f.ID_FACT
GROUP BY
  f.ID_FACT,
  f.ID_CLIENTE,
  c.NOMBRE,
  c.APELLIDO,
  c.TELEFONO,
  f.CANT_ART,
  f.ID_USUARIO,
  u.nombre,
  f.SUBTOTAL,
  f.FECHA_HORA,
  f.ID_TP
ORDER BY f.ID_FACT DESC

QUE HACE:
CONSULTA EL HISTORICO DE CORTES EN dbo.CORTE_DIARIO.
FILTRA LOS CORTES POR RANGO DE FECHAS CUANDO SE SELECCIONA DESDE Y HASTA.
SI RECIBE ID_USUARIO, FILTRA SOLO LOS CORTES DE ESE USUARIO.
CONSULTA LOS ABONOS EN dbo.PAGOS_CREDITO POR FECHA.
SI RECIBE ID_USUARIO, FILTRA SOLO LOS ABONOS DE ESE USUARIO.
GENERA UN CORTE NUEVO CALCULANDO VENTAS EN EFECTIVO, TRANSFERENCIA,
CREDITO, ABONOS DE CREDITO, TOTAL EN CAJA Y GANANCIA DEL DIA.
TAMBIEN CALCULA UN RESUMEN PREVIO SIN INSERTAR EN dbo.CORTE_DIARIO,
PARA QUE EL MODAL MUESTRE LOS DATOS DEL DIA AUNQUE NO EXISTA UN CORTE GUARDADO.
CUANDO SE CIERRA SESION, EL MODAL PIDE EL CONTEO FISICO DEL CAJERO
Y GUARDA ESE VALOR EN TOTAL_EN_CAJA.
EL CORTE GUARDADO USA ESTADO_CORTE = 4 PARA MARCARLO COMO CERRADO.
SI SE VUELVE A HACER CORTE EL MISMO DIA, SE INSERTA UN REGISTRO NUEVO.
CUANDO EL USUARIO INICIA SESION, EL MODAL DE INICIO DE CAJA PIDE EL MONTO
INICIAL Y LO GUARDA EN dbo.CORTE_DIARIO CON ESTADO_CORTE = 1.
LA COLUMNA FECHA_CORTE SE GUARDA CON FECHA Y HORA REALES DEL MOMENTO
EN QUE SE ABRE O CIERRA EL TURNO.
LOS CALCULOS DE CORTE USAN EL ULTIMO CORTE ABIERTO DEL DIA COMO DINERO_INICIA_CAJA.
AHORA EL TURNO SE MANEJA POR ID_USUARIO, NO SOLO POR FECHA.
EL LOGIN SOLO OMITE EL MODAL DE APERTURA SI ESE MISMO USUARIO YA TIENE
UN CORTE ABIERTO EL MISMO DIA.
ANTES DE INSERTAR EL CORTE CERRADO, EL PROCEDIMIENTO createDailyCut(date)
ACTUALIZA A ESTADO_CORTE = 4 CUALQUIER CORTE ABIERTO DEL MISMO DIA
Y DEL MISMO ID_USUARIO.
ESTA VALIDACION ES USADA DESDE src/app/app.ts EN shouldPromptOpeningCut(date),
DONDE SOLO SE OMITE EL MODAL DE INICIO DE CAJA SI EXISTE UN CORTE ABIERTO.

CONSULTA PARA CALCULAR VENTAS DEL CORTE:
WITH ventas AS (
  SELECT ve.PRECIO_VENTA * ve.CANT_PD AS total, ve.UTILIDAD AS utilidad, 1 AS id_tp, ve.FECHA_HORA, f.ID_USUARIO
  FROM dbo.VENTA_EFECTIVO ve
  INNER JOIN dbo.FACTURA f
    ON f.ID_FACT = ve.ID_FACT

  UNION ALL

  SELECT vc.PRECIO_VENTA * vc.CANT_PD AS total, vc.UTILIDAD AS utilidad, 2 AS id_tp, vc.FECHA_HORA, f.ID_USUARIO
  FROM dbo.VENTA_CREDITO vc
  INNER JOIN dbo.FACTURA f
    ON f.ID_FACT = vc.ID_FACT

  UNION ALL

  SELECT vt.PRECIO_VENTA * vt.CANT_PD AS total, vt.UTILIDAD AS utilidad, 3 AS id_tp, vt.FECHA_HORA, f.ID_USUARIO
  FROM dbo.VENTA_TRANSFERENCIA vt
  INNER JOIN dbo.FACTURA f
    ON f.ID_FACT = vt.ID_FACT
)
SELECT
  ISNULL(SUM(total), 0) AS venta_total_dia,
  ISNULL(SUM(CASE WHEN id_tp = 1 THEN total ELSE 0 END), 0) AS venta_efectivo,
  ISNULL(SUM(CASE WHEN id_tp = 2 THEN total ELSE 0 END), 0) AS venta_credito,
  ISNULL(SUM(CASE WHEN id_tp = 3 THEN total ELSE 0 END), 0) AS venta_transferencia,
  ISNULL(SUM(utilidad), 0) AS ganancia_del_dia,
  ISNULL((
    SELECT SUM(MONTO)
    FROM dbo.PAGOS_CREDITO
    WHERE TRY_CONVERT(date, FECHA_HORA) = TRY_CONVERT(date, @cut_date)
      AND ID_USUARIO = @user_id
  ), 0) AS abonos_credito
FROM ventas
WHERE TRY_CONVERT(date, FECHA_HORA) = TRY_CONVERT(date, @cut_date)
  AND ID_USUARIO = @user_id

UBICACION DEL CODIGO:

getProducts({ resolveProductImageUrl })

QUE HACE:
OBTIENE LOS PRODUCTOS ACTIVOS DESDE dbo.producto Y dbo.inventario.
ADEMAS CALCULA LA CANTIDAD VENDIDA DEL MES ANTERIOR SUMANDO
dbo.VENTA_EFECTIVO, dbo.VENTA_CREDITO Y dbo.VENTA_TRANSFERENCIA.
ESTA COLUMNA SE USA EN FACTURACION PARA MOSTRAR COMO PRIMERAS OPCIONES

CONSULTA SQL:
WITH ventas_mes_anterior AS (
  SELECT
    ID_PD,
    SUM(CANT_PD) AS CANTIDAD_VENDIDA_MES_ANTERIOR
  FROM (
    SELECT ID_PD, CANT_PD, TRY_CONVERT(datetime2, FECHA_HORA) AS sale_date
    FROM dbo.VENTA_EFECTIVO
    WHERE TRY_CONVERT(datetime2, FECHA_HORA) IS NOT NULL

    UNION ALL

    SELECT ID_PD, CANT_PD, TRY_CONVERT(datetime2, FECHA_HORA) AS sale_date
    FROM dbo.VENTA_CREDITO
    WHERE TRY_CONVERT(datetime2, FECHA_HORA) IS NOT NULL

    UNION ALL

    SELECT ID_PD, CANT_PD, TRY_CONVERT(datetime2, FECHA_HORA) AS sale_date
    FROM dbo.VENTA_TRANSFERENCIA
    WHERE TRY_CONVERT(datetime2, FECHA_HORA) IS NOT NULL
  ) ventas
  WHERE sale_date >= DATEADD(MONTH, DATEDIFF(MONTH, 0, GETDATE()) - 1, 0)
    AND sale_date < DATEADD(MONTH, DATEDIFF(MONTH, 0, GETDATE()), 0)
  GROUP BY ID_PD
)
SELECT
  p.id_producto,
  p.codigo,
  p.nombre,
  p.descripcion,
  p.imagen_url,
  i.precio_costo,
  i.precio_venta,
  i.precio_mayoreo,
  i.unidad_medida,
  i.stock,
  i.stock_minimo,
  i.stock_maximo,
  i.categoria,
  i.proveedor,
  i.creado_por,
  i.creado_en,
  i.actualizado_por,
  i.actualizado_en,
  ISNULL(vma.CANTIDAD_VENDIDA_MES_ANTERIOR, 0) AS CANTIDAD_VENDIDA_MES_ANTERIOR
FROM dbo.producto p
INNER JOIN dbo.inventario i
  ON i.codigo = p.codigo
LEFT JOIN ventas_mes_anterior vma
  ON vma.ID_PD = p.id_producto
WHERE p.activo = 1
ORDER BY p.nombre ASC;

RESPALDO DE CONSULTA PARA ALERTA DE CAIDA DE VENTAS

UBICACION DEL CODIGO:
server/data-access.js

PROCEDIMIENTO QUE LA UTILIZA:
getSalesDropAlert()

QUE HACE:
CONSULTA dbo.VENTA_EFECTIVO, dbo.VENTA_CREDITO Y dbo.VENTA_TRANSFERENCIA.
COMPARA LAS VENTAS DE HOY CONTRA LAS DE AYER A LA MISMA HORA.
LA ALERTA SOLO SE MARCA COMO ACTIVA A PARTIR DE LAS 07:00 PM.
SI HOY ESTA MAS BAJO QUE AYER, CALCULA LA BRECHA MONETARIA
Y EL PORCENTAJE DE CAIDA.

CONSULTA:
WITH ventas AS (
  SELECT TRY_CONVERT(datetime2, FECHA_HORA) AS sale_date, CAST(PRECIO_VENTA * CANT_PD AS decimal(18, 2)) AS total
  FROM dbo.VENTA_EFECTIVO
  WHERE TRY_CONVERT(datetime2, FECHA_HORA) IS NOT NULL

  UNION ALL

  SELECT TRY_CONVERT(datetime2, FECHA_HORA) AS sale_date, CAST(PRECIO_VENTA * CANT_PD AS decimal(18, 2)) AS total
  FROM dbo.VENTA_CREDITO
  WHERE TRY_CONVERT(datetime2, FECHA_HORA) IS NOT NULL

  UNION ALL

  SELECT TRY_CONVERT(datetime2, FECHA_HORA) AS sale_date, CAST(PRECIO_VENTA * CANT_PD AS decimal(18, 2)) AS total
  FROM dbo.VENTA_TRANSFERENCIA
  WHERE TRY_CONVERT(datetime2, FECHA_HORA) IS NOT NULL
),
limites AS (
  SELECT
    GETDATE() AS current_datetime,
    CAST(GETDATE() AS date) AS today_date
)
SELECT
  CAST(l.current_datetime AS datetime2) AS evaluated_at,
  CONVERT(varchar(8), CAST(l.current_datetime AS time), 108) AS cutoff_time,
  CASE
    WHEN CAST(l.current_datetime AS time) >= '19:00:00' THEN 1
    ELSE 0
  END AS is_active,
  ISNULL((
    SELECT SUM(v.total)
    FROM ventas v
    WHERE v.sale_date >= CAST(l.today_date AS datetime2)
      AND v.sale_date <= l.current_datetime
  ), 0) AS today_total,
  ISNULL((
    SELECT SUM(v.total)
    FROM ventas v
    WHERE v.sale_date >= DATEADD(DAY, -1, CAST(l.today_date AS datetime2))
      AND v.sale_date <= DATEADD(DAY, -1, l.current_datetime)
  ), 0) AS yesterday_total
FROM limites l

RESPALDO DE CONSULTA PARA ASISTENCIA Y PLANILLA CON USUARIOS REALES

UBICACION DEL CODIGO:
server/data-access.js

PROCEDIMIENTO QUE LA UTILIZA:
listAttendanceUsers()

QUE HACE:
CONSULTA dbo.usuario, dbo.ASISTENCIA Y dbo.VW_ASISTENCIA_HORAS.
TOMA EL REGISTRO DEL DIA ACTUAL PARA CADA USUARIO ACTIVO Y CALCULA
LAS HORAS TRABAJADAS DEL DIA Y LAS HORAS ACUMULADAS DE LOS ULTIMOS 7 DIAS.
SE USA PARA MOSTRAR DATOS REALES EN CONTROL DE ASISTENCIA Y PLANILLA.
TAMBIEN DEVUELVE EL HISTORIAL DIARIO PARA AGRUPARLO POR USUARIO, SEMANA Y DIA.
SI EXISTE LA COLUMNA dbo.ASISTENCIA.num_semana, EL HISTORIAL LA USA
PARA FILTRAR RAPIDAMENTE POR SEMANA DESDE EL FRONTEND.

CONSULTA:
WITH asistencia_hoy AS (
  SELECT
    a.ID_EMPLEADO,
    TRY_CONVERT(date, a.FECHA) AS FECHA_ASISTENCIA,
    CONVERT(varchar(5), TRY_CONVERT(time, a.HORA_ENTRADA), 108) AS HORA_ENTRADA,
    CONVERT(varchar(5), TRY_CONVERT(time, a.HORA_SALIDA), 108) AS HORA_SALIDA,
    UPPER(LTRIM(RTRIM(ISNULL(a.ESTADO, '')))) AS ESTADO,
    ROW_NUMBER() OVER (
      PARTITION BY a.ID_EMPLEADO
      ORDER BY a.ID_ASISTENCIA DESC
    ) AS rn
  FROM dbo.ASISTENCIA a
  WHERE TRY_CONVERT(date, a.FECHA) = CAST(GETDATE() AS date)
),
horas_hoy AS (
  SELECT
    vh.ID_EMPLEADO,
    SUM(CAST(ISNULL(vh.HORAS_TRABAJADAS, 0) AS decimal(18, 2))) AS HORAS_HOY
  FROM dbo.VW_ASISTENCIA_HORAS vh
  WHERE TRY_CONVERT(date, vh.FECHA) = CAST(GETDATE() AS date)
  GROUP BY vh.ID_EMPLEADO
),
horas_semana AS (
  SELECT
    vh.ID_EMPLEADO,
    SUM(CAST(ISNULL(vh.HORAS_TRABAJADAS, 0) AS decimal(18, 2))) AS HORAS_SEMANA,
    COUNT(DISTINCT TRY_CONVERT(date, vh.FECHA)) AS DIAS_TRABAJADOS
  FROM dbo.VW_ASISTENCIA_HORAS vh
  WHERE TRY_CONVERT(date, vh.FECHA) BETWEEN DATEADD(day, -6, CAST(GETDATE() AS date)) AND CAST(GETDATE() AS date)
  GROUP BY vh.ID_EMPLEADO
)
SELECT
  u.id_usuario,
  u.nombre,
  u.usuario,
  u.rol,
  CAST(GETDATE() AS date) AS FECHA_REFERENCIA,
  ah.FECHA_ASISTENCIA,
  ah.HORA_ENTRADA,
  ah.HORA_SALIDA,
  CASE
    WHEN ah.ESTADO IN ('AUSENTE', 'PERMISO', 'VACACIONES', 'INCAPACIDAD') THEN ah.ESTADO
    WHEN ISNULL(hh.HORAS_HOY, 0) > 0 AND ISNULL(ah.HORA_ENTRADA, '00:00') > '08:00' THEN 'TARDE'
    WHEN ISNULL(hh.HORAS_HOY, 0) > 0 THEN 'PUNTUAL'
    ELSE 'SIN REGISTRO'
  END AS ESTADO_ASISTENCIA,
  CAST(ISNULL(hh.HORAS_HOY, 0) AS decimal(18, 2)) AS HORAS_HOY,
  CAST(ISNULL(hs.HORAS_SEMANA, 0) AS decimal(18, 2)) AS HORAS_SEMANA,
  CAST(ISNULL(hs.DIAS_TRABAJADOS, 0) AS int) AS DIAS_TRABAJADOS
FROM dbo.usuario u
LEFT JOIN asistencia_hoy ah
  ON ah.ID_EMPLEADO = u.id_usuario
  AND ah.rn = 1
LEFT JOIN horas_hoy hh
  ON hh.ID_EMPLEADO = u.id_usuario
LEFT JOIN horas_semana hs
  ON hs.ID_EMPLEADO = u.id_usuario
WHERE u.activo = 1
ORDER BY u.nombre ASC

CONSULTA DE HISTORIAL:
WITH asistencia_historial AS (
  SELECT
    a.ID_ASISTENCIA,
    a.ID_EMPLEADO,
    TRY_CONVERT(date, a.FECHA) AS FECHA_ASISTENCIA,
    CONVERT(varchar(5), TRY_CONVERT(time, a.HORA_ENTRADA), 108) AS HORA_ENTRADA,
    CONVERT(varchar(5), TRY_CONVERT(time, a.HORA_SALIDA), 108) AS HORA_SALIDA,
    UPPER(LTRIM(RTRIM(ISNULL(a.ESTADO, '')))) AS ESTADO,
    ROW_NUMBER() OVER (
      PARTITION BY a.ID_EMPLEADO, TRY_CONVERT(date, a.FECHA)
      ORDER BY a.ID_ASISTENCIA DESC
    ) AS rn
  FROM dbo.ASISTENCIA a
  WHERE TRY_CONVERT(date, a.FECHA) >= DATEADD(day, -84, CAST(GETDATE() AS date))
)
SELECT
  u.id_usuario,
  ah.ID_ASISTENCIA,
  ah.FECHA_ASISTENCIA,
  ISNULL(ah.num_semana, DATEPART(ISO_WEEK, ah.FECHA_ASISTENCIA)) AS NUM_SEMANA,
  ah.HORA_ENTRADA,
  ah.HORA_SALIDA,
  ah.ESTADO,
  CAST(ISNULL(vh.HORAS_TRABAJADAS, 0) AS decimal(18, 2)) AS HORAS_TRABAJADAS
FROM dbo.usuario u
LEFT JOIN asistencia_historial ah
  ON ah.ID_EMPLEADO = u.id_usuario
  AND ah.rn = 1
LEFT JOIN dbo.VW_ASISTENCIA_HORAS vh
  ON vh.ID_EMPLEADO = ah.ID_EMPLEADO
  AND TRY_CONVERT(date, vh.FECHA) = ah.FECHA_ASISTENCIA
WHERE u.activo = 1
  AND ah.FECHA_ASISTENCIA IS NOT NULL
ORDER BY u.nombre ASC, ah.FECHA_ASISTENCIA DESC

RESPALDO DE CONSULTA PARA GUARDAR MARCAS DE ASISTENCIA

UBICACION DEL CODIGO:
server/data-access.js

PROCEDIMIENTO QUE LA UTILIZA:
saveAttendanceMark()

QUE HACE:
CREA O ACTUALIZA LA MARCA DIARIA EN dbo.ASISTENCIA
USANDO FECHA, HORA_ENTRADA Y HORA_SALIDA EN FORMATO MILITAR HH:mm.
SI YA EXISTE UNA FILA PARA EL USUARIO Y LA FECHA, LA ACTUALIZA.
SI NO EXISTE, CREA UNA NUEVA FILA.
TAMBIEN LLENA dbo.ASISTENCIA.num_semana CON DATEPART(ISO_WEEK, @attendance_date)
PARA QUE ASISTENCIA Y PLANILLA PUEDAN FILTRAR RAPIDAMENTE POR SEMANA.

CONSULTA:
IF EXISTS (
  SELECT 1
  FROM dbo.ASISTENCIA
  WHERE ID_EMPLEADO = @employee_id
    AND FECHA = @attendance_date
)
BEGIN
  UPDATE dbo.ASISTENCIA
  SET HORA_ENTRADA = CONVERT(datetime, @attendance_date_text + 'T' + @entry_time_text + ':00', 126),
      HORA_SALIDA = CONVERT(datetime, @attendance_date_text + 'T' + @exit_time_text + ':00', 126),
      num_semana = DATEPART(ISO_WEEK, @attendance_date),
      ESTADO = @attendance_status,
      OBSERVACION = @observation,
      USUARIO_REGISTRO = @recorded_by,
      FECHA_REGISTRO = GETDATE()
  WHERE ID_EMPLEADO = @employee_id
    AND FECHA = @attendance_date;
END
ELSE
BEGIN
  INSERT INTO dbo.ASISTENCIA (
    ID_EMPLEADO,
    FECHA,
    HORA_ENTRADA,
    HORA_SALIDA,
    num_semana,
    ESTADO,
    OBSERVACION,
    USUARIO_REGISTRO,
    FECHA_REGISTRO
  )
  VALUES (
    @employee_id,
    @attendance_date,
    CONVERT(datetime, @attendance_date_text + 'T' + @entry_time_text + ':00', 126),
    CONVERT(datetime, @attendance_date_text + 'T' + @exit_time_text + ':00', 126),
    DATEPART(ISO_WEEK, @attendance_date),
    @attendance_status,
  @observation,
  @recorded_by,
    GETDATE()
  );
END
